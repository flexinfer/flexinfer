QUICK START:

1. Create your first model:

   kubectl apply -f - <<EOF
   apiVersion: ai.flexinfer.io/v1alpha2
   kind: Model
   metadata:
     name: llama3
     namespace: {{ .Release.Namespace }}
   spec:
     backend: ollama
     source: ollama://llama3.2:1b
     gpu:
       count: 1
   EOF

2. Wait for the model to be ready:

   kubectl wait --for=condition=Ready model/llama3 -n {{ .Release.Namespace }} --timeout=300s

3. Access the inference API:

{{- if .Values.proxy.enabled }}
   # Port-forward to the proxy service
   kubectl port-forward svc/{{ include "flexinfer.fullname" . }}-proxy 8080:{{ .Values.proxy.service.port }} -n {{ .Release.Namespace }}

   # Send a request
   curl http://localhost:8080/v1/chat/completions \
     -H "Content-Type: application/json" \
     -d '{"model": "llama3", "messages": [{"role": "user", "content": "Hello!"}]}'
{{- else }}
   # Direct access to model endpoint (check model status for endpoint)
   kubectl get model llama3 -n {{ .Release.Namespace }} -o jsonpath='{.status.endpoint}'
{{- end }}
